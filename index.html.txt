    <script>
        let balance = parseInt(localStorage.getItem('user_balance')) || 100000;
        let userName = localStorage.getItem('user_name') || "ThànhVien_" + Math.floor(100 + Math.random()*900);
        let userAcc = localStorage.getItem('user_acc') || "player" + Math.floor(1000 + Math.random()*9000);
        
        localStorage.setItem('user_balance', balance);
        localStorage.setItem('user_name', userName);
        localStorage.setItem('user_acc', userAcc);

        document.getElementById('userName').innerText = userName;
        document.getElementById('userAccDisplay').innerText = userAcc;
        document.getElementById('userAvatar').src = `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`;
        document.getElementById('balanceDisplay').innerText = balance.toLocaleString() + " VNĐ";

        let timeLeft = 30;
        let currentBet = null;
        let cauHistory = JSON.parse(localStorage.getItem('tx_cau')) || [];

        setInterval(() => {
            timeLeft--;
            document.getElementById('timerDisplay').innerText = `Thời gian: ${timeLeft}s`;

            if (timeLeft <= 0) {
                timeLeft = 30;
                rollDiceAndResult();
            }
        }, 1000);

        function rollDiceAndResult() {
            let d1 = Math.floor(Math.random() * 6) + 1;
            let d2 = Math.floor(Math.random() * 6) + 1;
            let d3 = Math.floor(Math.random() * 6) + 1;
            let total = d1 + d2 + d3;
            let result = total >= 11 ? "TÀI" : "XỈU";
            let session = "#" + Math.floor(100000 + Math.random() * 900000);

            document.getElementById('diceArea').innerHTML = getDiceHTML(d1) + getDiceHTML(d2) + getDiceHTML(d3);
            document.getElementById('sessionId').innerText = session;
            addLog(`Kết quả: ${d1}-${d2}-${d3} = ${total} (${result})`, "text-blue-400");

            cauHistory.unshift({ session, result, detail: `${d1}+${d2}+${d3}=${total}` });
            if (cauHistory.length > 30) cauHistory.pop();
            localStorage.setItem('tx_cau', JSON.stringify(cauHistory));

            // Lưu dữ liệu để trang Admin đọc được
            let myStats = JSON.parse(localStorage.getItem('tx_my_stats')) || { name: userName, acc: userAcc, win: 0 };
            
            if (currentBet) {
                if (currentBet.choice === result) {
                    balance += currentBet.amount;
                    myStats.win += currentBet.amount;
                    addLog(`🎉 Thắng +${currentBet.amount.toLocaleString()}đ`, "text-emerald-400 font-bold");
                } else {
                    balance -= currentBet.amount;
                    addLog(`❌ Thua -${currentBet.amount.toLocaleString()}đ`, "text-rose-400");
                }
                localStorage.setItem('user_balance', balance);
                document.getElementById('balanceDisplay').innerText = balance.toLocaleString() + " VNĐ";
                currentBet = null;
            }
            myStats.balance = balance;
            localStorage.setItem('tx_my_stats', JSON.stringify(myStats));
        }

        function placeBet(choice) {
            if (timeLeft <= 5) return alert("Sắp hết giờ, không thể đặt cược phiên này!");
            let amt = parseInt(document.getElementById('betAmount').value);
            if (isNaN(amt) || amt <= 0) return alert("Số tiền cược không hợp lệ!");
            if (amt > balance) return alert("Số dư không đủ!");

            currentBet = { choice, amount: amt };
            addLog(`Đã cược ${choice} số tiền ${amt.toLocaleString()}đ`, "text-amber-400");
        }

        function submitWithdraw() {
            let amt = parseInt(document.getElementById('withdrawAmount').value);
            let bank = document.getElementById('withdrawBank').value.trim();
            if (isNaN(amt) || amt <= 0 || !bank) return alert("Vui lòng nhập đủ thông tin rút tiền!");
            if (amt > balance) return alert("Số dư không đủ để rút!");

            balance -= amt;
            localStorage.setItem('user_balance', balance);
            document.getElementById('balanceDisplay').innerText = balance.toLocaleString() + " VNĐ";

            let reqs = JSON.parse(localStorage.getItem('tx_withdraws')) || [];
            reqs.unshift({ user: userName, acc: userAcc, amount: amt, bank, status: 'Chờ duyệt' });
            localStorage.setItem('tx_withdraws', JSON.stringify(reqs));

            closeModal('withdrawModal');
            alert("Gửi lệnh rút tiền thành công!");
        }

        function getDiceHTML(num) {
            let dots = {1:[[2,2]], 2:[[1,1],[3,3]], 3:[[1,1],[2,2],[3,3]], 4:[[1,1],[1,3],[3,1],[3,3]], 5:[[1,1],[1,3],[2,2],[3,1],[3,3]], 6:[[1,1],[1,3],[2,1],[2,3],[3,1],[3,3]]};
            let html = '';
            (dots[num] || [[2,2]]).forEach(p => html += `<div class="dot" style="grid-area: ${p[0]} / ${p[1]};"></div>`);
            return `<div class="dice-box">${html}</div>`;
        }

        function addLog(msg, color) {
            let logs = document.getElementById('logs');
            let p = document.createElement('p');
            p.className = color;
            p.innerText = `[${new Date().toLocaleTimeString()}] ${msg}`;
            logs.appendChild(p);
            logs.scrollTop = logs.scrollHeight;
        }

        function openModal(id) {
            if (id === 'cauModal') {
                document.getElementById('cauList').innerHTML = cauHistory.map(i => `<div class="flex justify-between bg-slate-800 p-2 rounded"><span>${i.session} (${i.detail})</span><b class="${i.result==='TÀI'?'text-rose-500':'text-blue-500'}">${i.result}</b></div>`).join('');
            }
            if (id === 'topModal') {
                let stats = JSON.parse(localStorage.getItem('tx_my_stats')) || { name: userName, win: 0 };
                document.getElementById('topList').innerHTML = `<div class="flex justify-between bg-slate-800 p-2 rounded"><span class="text-amber-400">#1 ${stats.name}</span><span class="text-amber-400">+${(stats.win||0).toLocaleString()}đ</span></div>`;
            }
            document.getElementById(id).classList.add('active');
        }
        function closeModal(id) { document.getElementById(id).classList.remove('active'); }
    </script>
